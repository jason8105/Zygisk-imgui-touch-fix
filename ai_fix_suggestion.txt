#include <unistd.h>
#include "imgui.h"
#include "zygisk.hpp"

#define LOG_TAG "ZygiskImGui"
#define LOGI(...) __android_log_print(ANDROID_LOG_INFO, LOG_TAG, __VA_ARGS__)
#define LOGE(...) __android_log_print(ANDROID_LOG_ERROR, LOG_TAG, __VA_ARGS__)

// Universal Touch Hook Implementation via AInputQueue or native event dispatch
typedef int (*AInputQueue_preDispatchEvent_t)(AInputQueue* queue, AInputEvent* event);
static AInputQueue_preDispatchEvent_t original_AInputQueue_preDispatchEvent = nullptr;

int hooked_AInputQueue_preDispatchEvent(AInputQueue* queue, AInputEvent* event) {
    if (event && AInputEvent_getType(event) == AINPUT_EVENT_TYPE_MOTION) {
        int32_t action = AMotionEvent_getAction(event);
        int32_t actionMasked = action & AMOTION_EVENT_ACTION_MASK;
        float x = AMotionEvent_getX(event, 0);
        float y = AMotionEvent_getY(event, 0);

        auto& io = ImGui::GetIO();
        io.AddMousePosEvent(x, y);

        if (actionMasked == AMOTION_EVENT_ACTION_DOWN || actionMasked == AMOTION_EVENT_ACTION_POINTER_DOWN) {
            io.AddMouseButtonEvent(ImGuiMouseButton_Left, true);
        } else if (actionMasked == AMOTION_EVENT_ACTION_UP || actionMasked == AMOTION_EVENT_ACTION_POINTER_UP) {
            io.AddMouseButtonEvent(ImGuiMouseButton_Left, false);
        }

        if (io.WantCaptureMouse) {
            // Consume event if ImGui captures touch
            return 1;
        }
    }
    if (original_AInputQueue_preDispatchEvent) {
        return original_AInputQueue_preDispatchEvent(queue, event);
    }
    return 0;
}

void* hook_thread(void*) {
    LOGI("Initializing Universal ImGui Touch Fix Zygisk Module...");
    ImGui::CreateContext();

    // Hook AInputQueue_preDispatchEvent dynamically
    void* handle = dlopen("libandroid.so", RTLD_GLOBAL | RTLD_LAZY);
    if (handle) {
        auto fn = (AInputQueue_preDispatchEvent_t)dlsym(handle, "AInputQueue_preDispatchEvent");
        if (fn) {
            original_AInputQueue_preDispatchEvent = fn;
            LOGI("Successfully hooked AInputQueue_preDispatchEvent for universal touch!");
        }
    }
    return nullptr;
}

class ZygiskModule : public zygisk::ModuleBase {

    }
