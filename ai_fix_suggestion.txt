    ImGuiIO& io = ImGui::GetIO();
    io.BackendPlatformName = "imgui_impl_android";
    return true;

    ImGuiIO& io = ImGui::GetIO();
    io.BackendPlatformName = NULL;
}

void ImGui_ImplAndroid_NewFrame() {
    ImGuiIO& io = ImGui::GetIO();

    struct timespec res;
    clock_gettime(CLOCK_MONOTONIC, &res);
    double current_time = (double)res.tv_sec + (double)res.tv_nsec / 1000000000.0;
    if (g_Time <= 0.0)
        g_Time = current_time;
    io.DeltaTime = (float)(current_time - g_Time);
    g_Time = current_time;
}

int hooked_AInputQueue_preDispatchEvent(AInputQueue* queue, AInputEvent* event) {

    }
    return orig_AInputQueue_preDispatchEvent ? orig_AInputQueue_preDispatchEvent(queue, event) : 0;

bool HandleTouchInput(AInputEvent* event) {
    int32_t type = AInputEvent_getType(event);
    if (type == AINPUT_EVENT_TYPE_MOTION) {
        int32_t action = AMotionEvent_getAction(event);
        int32_t maskedAction = action & AMOTION_EVENT_ACTION_MASK;
        
        float x = AMotionEvent_getX(event, 0);
        float y = AMotionEvent_getY(event, 0);

        ImGuiIO& io = ImGui::GetIO();

        if (maskedAction == AMOTION_EVENT_ACTION_DOWN || maskedAction == AMOTION_EVENT_ACTION_MOVE || maskedAction == AMOTION_EVENT_ACTION_POINTER_DOWN) {
            io.AddMousePosEvent(x, y);
            io.AddMouseButtonEvent(0, true);
        } else if (maskedAction == AMOTION_EVENT_ACTION_UP || maskedAction == AMOTION_EVENT_ACTION_POINTER_UP || maskedAction == AMOTION_EVENT_ACTION_CANCEL) {
            io.AddMousePosEvent(x, y);
            io.AddMouseButtonEvent(0, false);
        }

        return io.WantCaptureMouse;
    }
    return false;
}

#include "zygisk.hpp"

#define TAG "ZygiskImGui"

class MyModule : public zygisk::ModuleBase {
public:
    void onLoad(zygisk::Api *api, JNIEnv *env) override {
        this->api = api;
        this->env = env;
        LOGI("Zygisk module loaded successfully for Magisk 24-26!");
    }

    void preAppSpecialize(zygisk::AppSpecializeArgs *args) override {
        // Safe pre-app specialization for Magisk v24-26
    }

    void postAppSpecialize(zygisk::AppSpecializeArgs *args) override {
        IMGUI_CHECKVERSION();
        ImGui::CreateContext();
        ImGuiIO& io = ImGui::GetIO(); (void)io;

        InstallUniversalTouchHook();
        LOGI("Post app specialize complete. ImGui context created.");
    }

private:
    zygisk::Api *api;
    JNIEnv *env;
};

REGISTER_ZYGISK_MODULE(MyModule)

=== FILE: make_module.sh
#!/usr/bin/env bash
set -e

echo "Building native libraries..."
./gradlew :app:assembleRelease

echo "Assembling Magisk Module Zip..."
rm -rf build_module
mkdir -p build_module/zygisk/arm64-v8a
mkdir -p build_module/zygisk/armeabi-v7a

# Copy built .so files from outputs to Magisk module layout
if [ -f "app/build/intermediates/cmake/release/obj/arm64-v8a/libzygisk_mod.so" ]; then
    cp app/build/intermediates/cmake/release/obj/arm64-v8a/libzygisk_mod.so build_module/zygisk/arm64-v8a/libzygisk.so
elif [ -f "app/build/outputs/cmake/release/obj/arm64-v8a/libzygisk_mod.so" ]; then
    cp app/build/outputs/cmake/release/obj/arm64-v8a/libzygisk_mod.so build_module/zygisk/arm64-v8a/libzygisk.so
fi

if [ -f "app/build/intermediates/cmake/release/obj/armeabi-v7a/libzygisk_mod.so" ]; then
    cp app/build/intermediates/cmake/release/obj/armeabi-v7a/libzygisk_mod.so build_module/zygisk/armeabi-v7a/libzygisk.so
elif [ -f "app/build/outputs/cmake/release/obj/armeabi-v7a/libzygisk_mod.so" ]; then
    cp app/build/outputs/cmake/release/obj/armeabi-v7a/libzygisk_mod.so build_module/zygisk/armeabi-v7a/libzygisk.so
fi

cp module.prop build_module/
touch build_module/customize.sh

cd build_module
zip -r ../zygisk_imgui_release.zip *
cd ..

echo "Module successfully packaged as zygisk_imgui_release.zip!"
