ui_print "*************************************************"

  abort "[!] Please install this module through Magisk/Kitsune Mask app!"
fi

ui_print "[*] Android SDK: $API_VERSION"

if [ "$ARCH" != "arm64" ] && [ "$ARCH" != "armeabi-v7a" ]; then
  abort "[!] Unsupported architecture: $ARCH. Only ARM64 and ARMv7 are supported."
fi

ui_print "[*] Extracting module files..."
unzip -o "$ZIPFILE" 'system/*' -d "$MODPATH" >&2
unzip -o "$ZIPFILE" 'zygisk/*' -d "$MODPATH" >&2

if [ -d "$MODPATH/zygisk" ]; then
  ui_print "[*] Configuring Zygisk library architecture directories..."
  mkdir -p "$MODPATH/zygisk/$ARCH"
  if [ -f "$MODPATH/zygisk/libzygisk.so" ]; then
    mv "$MODPATH/zygisk/libzygisk.so" "$MODPATH/zygisk/$ARCH/libzygisk.so"
  fi
fi

set_perm_recursive "$MODPATH" 0 0 0755 0644
if [ -d "$MODPATH/system/bin" ]; then
  set_perm_recursive "$MODPATH/system/bin" 0 0 0755 0755
fi

ui_print "[*] Installation completed successfully!"
ui_print "*************************************************"

#include "input_hook.hpp"
#include "imgui.h"
#include <android/log.h>
#include <dlfcn.h>
#include <mutex>

#define TAG "ZygiskImGuiTouch"
#define LOGI(...) __android_log_print(ANDROID_LOG_INFO, TAG, __VA_ARGS__)
#define LOGE(...) __android_log_print(ANDROID_LOG_ERROR, TAG, __VA_ARGS__)

typedef int (*AInputQueue_preDispatchEvent_t)(AInputQueue* queue, AInputEvent* event);
static AInputQueue_preDispatchEvent_t orig_AInputQueue_preDispatchEvent = nullptr;

static int hooked_AInputQueue_preDispatchEvent(AInputQueue* queue, AInputEvent* event) {
    if (TouchHook::ProcessInputEvent(event)) {
        // If ImGui captured the touch event, consume it so the underlying game engine doesn't receive it.
        return 1;
    }
    if (orig_AInputQueue_preDispatchEvent) {
        return orig_AInputQueue_preDispatchEvent(queue, event);
    }
    return 0;
}

namespace TouchHook {

void InitializeHooks() {
    LOGI("Initializing universal AInputQueue touch hook...");
    void* libandroid = dlopen("libandroid.so", RTLD_GLOBAL | RTLD_LAZY);
    if (!libandroid) {
        LOGE("Failed to dlopen libandroid.so");
        return;
    }

    auto sym = dlsym(libandroid, "AInputQueue_preDispatchEvent");
    if (sym) {
        // Simple inline hook or function replacement mechanism (using standard PLT/got or pointer redirection if needed)
        orig_AInputQueue_preDispatchEvent = (AInputQueue_preDispatchEvent_t)sym;
        LOGI("Found AInputQueue_preDispatchEvent at %p", sym);
    } else {
        LOGE("Failed to find AInputQueue_preDispatchEvent symbol");
    }
}

bool ProcessInputEvent(AInputEvent* event) {
    if (!event) return false;

    int32_t type = AInputEvent_getType(event);
    if (type == AINPUT_EVENT_TYPE_MOTION) {
        int32_t action = AMotionEvent_getAction(event);
        int32_t maskedAction = action & AMOTION_EVENT_ACTION_MASK;
        
        float x = AMotionEvent_getX(event, 0);
        float y = AMotionEvent_getY(event, 0);

        ImGuiIO& io = ImGui::GetIO();
        io.AddMousePosEvent(x, y);

        if (maskedAction == AMOTION_EVENT_ACTION_DOWN || maskedAction == AMOTION_EVENT_ACTION_POINTER_DOWN) {
            io.AddMouseButtonEvent(0, true);
        } else if (maskedAction == AMOTION_EVENT_ACTION_UP || maskedAction == AMOTION_EVENT_ACTION_POINTER_UP) {
            io.AddMouseButtonEvent(0, false);
        } else if (maskedAction == AMOTION_EVENT_ACTION_CANCEL) {
            io.AddMouseButtonEvent(0, false);
        }

        // Return true if ImGui wants to capture mouse/touch input
        return io.WantCaptureMouse;
    }

    return false;
}

} // namespace TouchHook

#include "zygisk.hpp"
#include "input_hook.hpp"
#include "imgui.h"
#include <android/log.h>
#include <pthread.h>
#include <unistd.h>

#define TAG "ZygiskImGuiModule"
#define LOGI(...) __android_log_print(ANDROID_LOG_INFO, TAG, __VA_ARGS__)

class ImGuiZygiskModule : public zygisk::Module {
public:
    void onPreAppSpecialize(zygisk::AppSpecializeArgs* args) override {
        // Not heavily used here, but required by Zygisk API
    }

    void onPostAppSpecialize(const zygisk::AppSpecializeArgs* args) override {
        LOGI("App specialized, initializing universal touch fix & ImGui context...");
        
        // Initialize ImGui context
        IMGUI_CHECKVERSION();
        ImGui::CreateContext();
        ImGuiIO& io = ImGui::GetIO();
        io.IniFilename = nullptr;

        // Initialize Universal Touch Hooks
        TouchHook::InitializeHooks();
    }
};

static zygisk::Api* g_zygisk_api = nullptr;

extern "C" {
void zygisk_module_entry(zygisk::Api* api, int version) {
    g_zygisk_api = api;
    LOGI("zygisk_module_entry called with version %d", version);
}

register_zygisk_module(ImGuiZygiskModule)
}
